cmake_minimum_required(VERSION 3.16)
project(linux-mount-auditor VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mount_auditor_lib
  src/util.cpp src/mountinfo.cpp src/fstab.cpp src/model.cpp src/render.cpp)
target_include_directories(mount_auditor_lib PUBLIC include)
target_compile_options(mount_auditor_lib PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  target_compile_options(mount_auditor_lib PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mount_auditor_lib PRIVATE -fsanitize=address,undefined)
endif()
add_executable(linux-mount-auditor src/main.cpp)
target_link_libraries(linux-mount-auditor PRIVATE mount_auditor_lib)
target_compile_options(linux-mount-auditor PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  target_compile_options(linux-mount-auditor PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(linux-mount-auditor PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS linux-mount-auditor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/linux-mount-auditor.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
if(BUILD_TESTING)
  add_executable(unit-tests tests/test_main.cpp)
  target_link_libraries(unit-tests PRIVATE mount_auditor_lib)
  target_compile_options(unit-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(unit-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(unit-tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME unit COMMAND unit-tests)
  add_test(NAME cli-help COMMAND linux-mount-auditor help)
  add_test(NAME cli-version COMMAND linux-mount-auditor version)
  add_test(NAME cli-fixture-json COMMAND linux-mount-auditor inspect --mountinfo ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/mountinfo.txt --json)
  add_test(NAME cli-suite COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.sh $<TARGET_FILE:linux-mount-auditor> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
set(CPACK_PACKAGE_NAME "linux-mount-auditor")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_PACKAGE_CONTACT "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/guhshosting/linux-mount-auditor")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_DEBIAN_PACKAGE_DESCRIPTION "Read-only Linux mount configuration auditor")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_GROUP "Applications/System")
set(CPACK_RPM_PACKAGE_URL "https://github.com/guhshosting/linux-mount-auditor")
include(CPack)
